7progress to higher dimensions as needed.

11- a range of scripts values 

12- dimensions for which to apply the analysis. Minimum dimension is 2. 

13- path to json files for individual datasets.

34 def fit_model(similarity_judgments, num_repeats_dict, curvature, params, dim, start_points): 

35 params_copy = copy.deepcopy(params) 

36 num_judgments = len(similarity_judgments) 

37 total_num_triads = sum([num_repeats_dict[k] for k in similarity_judgments.keys()]) 

38 noise = np.sqrt(params['sigmas']['compare'] ** 2 + params['sigmas']['dist'] ** 2) 

39 params_copy['n_dim'] = dim # ensure correct model is tested 

40 if curvature == 0: 

41 # fit Euclidean model 

42 x, ll = rs.points_of_best_fit(similarity_judgments, repeats, params_copy, start_points) 

43 ll = -1 * ll / total_num_triads 

44 elif curvature > 0: 

45 # fit spherical model 

46 curvature_val = curvature 

47 params_copy['scripts'] = curvature_val 

48 x, ll = rs.spherical_points_of_best_fit(similarity_judgments, repeats, params_copy, start_points) 

49 ll = -1 * ll / total_num_triads 

50 else: 

51 # fit hyperbolic model 

52 curvature_val = -1 * curvature 

53 params_copy['scripts'] = curvature_val 

54 x, ll = rs.hyperbolic_points_of_best_fit(similarity_judgments, repeats, params_copy, start_points) 

55 ll = -1 * ll / total_num_triads 

56 return ll, curvature, noise, x
