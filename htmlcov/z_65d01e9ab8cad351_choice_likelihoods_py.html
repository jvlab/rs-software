1""" 

2Utilities for computing log-likelihoods for geometry-based comparison data. 

3 

4This file provides methods for evaluating the log-likelihood of aggregated 

5comparison counts under a distance-geometry response model. The observed data 

6are of the form $N((i, j) > (k, l))$, meaning the number of times one pair of 

7stimuli was judged more different than another pair. 

8 

9The model computes choice probabilities from interstimulus distances using an 

10error-function approximation, then evaluates the corresponding log-likelihood. 

11It also includes reference calculations for random-choice and empirical 

12best-case benchmarks. 

13 

14Note: 

15The likelihood calculations use log2, so the returned values are in bits. 

16When `sigma_point = 0`, the Gaussian-noise assumption matches the erf-based 

17probability exactly. When `sigma_point > 0`, the `erf` probability is only an 

18approximation because the full noise model is not strictly Gaussian. 

19(By `sigma_point`, we mean the standard deviation of additive noise on stimulus 

20 coordinates before distances are computed.) 

21""" 

22

35def calculate_ll(counts, probs, num_repeats): 

36 """ 

37 Calculate the log-likelihood of observed counts given model probabilities. 

38 

39 This function augments each comparison with its reverse outcome, checks whether 

40 the model assigns zero probability to any observed event, and returns the 

41 log-likelihood using a base-2 logarithm. 

42 

43 Args: 

44 counts (numpy.ndarray): Observed counts for each comparison direction. 

45 probs (numpy.ndarray): Model probabilities for the corresponding 

46 comparisons. 

47 num_repeats (numpy.ndarray): Total number of repeats for each comparison. 

48 

49 Returns: 

50 tuple[float, bool]: The log-likelihood and a boolean flag indicating whether the model is infeasible. 

51 The flag is `True` when an observed event has zero model probability. 

52 

53 Note: 

54 The function uses a variable called `EPSILON` set to `1e-30.` This very small positive constant is added only when a model assigns 

55 zero probability to an observed comparison. It prevents `log2(0)` from causing an error 

56 during likelihood computation, while keeping the adjustment negligible. 

57 """ 

58 

59 reverse_counts = num_repeats - counts 

60 reverse_probs = 1 - probs 

61 

62 probs = concatenate((probs, reverse_probs)) 

63 counts = concatenate((counts, reverse_counts)) 

64 

65 model_bad = False 

66 # check if geometry is bad, i.e. prob = 0 but count > 0 

67 prob_zero = probs == 0 

68 if (counts[prob_zero] > 0).any(): 

69 model_bad = True 

70 

71 # make sure there are no zero probabilities (avoid log(0) error) 

72 probs[prob_zero] += EPSILON 

73 log_likelihood = counts.dot(log2(probs)) 

74 return log_likelihood, model_bad

77def dist_model_ll_vectorized(pair_a, pair_b, judgment_counts, judgment_repeats, params, stimuli): 

78 """ 

79 Compute the log-likelihood under a specific perceptual space model with given distances between points. 

80 

81 This function converts stimulus coordinates into a full pairwise distance 

82 matrix, computes comparison probabilities with `find_probabilities`, and then 

83 evaluates the log-likelihood against the observed judgment counts. 

84 

85 Args: 

86 pair_a (numpy.ndarray): Array of index pairs defining the first distance in 

87 each comparison. 

88 pair_b (numpy.ndarray): Array of index pairs defining the second distance 

89 in each comparison. 

90 judgment_counts (numpy.ndarray): Observed counts for the first pair being 

91 judged more dissimilar than the second. 

92 judgment_repeats (numpy.ndarray): Total number of repeats for each 

93 comparison. 

94 params (dict): Model parameters. Must contain `noise_st_dev`. 

95 stimuli (numpy.ndarray): Stimulus coordinates used to compute pairwise 

96 distances. 

97 

98 Returns: 

99 tuple[float, bool]: The log-likelihood and the infeasibility flag returned by `calculate_ll`. 

100 

101 See Also: 

102 find_probabilities: Convert distance differences into choice probabilities. 

103 """ 

104 # get geometry probabilities and join counts and geometry prob for each trial (N, p) 

105 interstimulus_distances = squareform(pdist(stimuli)) 

106 probs = find_probabilities(interstimulus_distances, pair_a, pair_b, params['noise_st_dev']) 

107 # calculate log-likelihood, is_bad flag 

108 return calculate_ll(judgment_counts, probs, judgment_repeats)

111def find_probabilities(distances, pair_a, pair_b, noise_st_dev): 

112 """ 

113 Compute choice probabilities from pairwise distance differences. 

114 

115 For each comparison, this function subtracts the second pair distance from the 

116 first pair distance and converts that difference into a probability that the 

117 first pair is judged more different. 

118 

119 Args: 

120 distances (numpy.ndarray): Square matrix of pairwise stimulus distances. 

121 pair_a (numpy.ndarray): Array of index pairs defining the first distance in 

122 each comparison. 

123 pair_b (numpy.ndarray): Array of index pairs defining the second distance in each comparison. 

124 noise_st_dev (float): Standard deviation of the combined Gaussian noise term. 

125 

126 Returns: 

127 numpy.ndarray: Probability that the first pair is judged more different than the second. 

128 

129 Note: 

130 When `noise_st_dev = 0`, the output is deterministic, with probabilities `0`, `0.5`, or `1` 

131 depending on the sign of the distance difference. When `noise_st_dev > 0`, the probability 

132 is computed using an approximation based on the erf function. 

133 """ 

134 difference = distances[pair_a[:, 0], pair_a[:, 1]] - distances[pair_b[:, 0], pair_b[:, 1]] 

135 if noise_st_dev == 0: 

136 # if (sigmas['compare'] + sigmas['dist'] == 0) or no_noise is True: 

137 probabilities = (difference < 0) * 0 + (difference > 0) * 1 + (difference == 0) * 0.5 

138 else: 

139 # total_st_dev = sqrt((sigmas['dist'] ** 2) + sigmas['compare'] ** 2) 

140 probabilities = 0.5 * (1 + erf(difference / float(2 * noise_st_dev))) 

141 return probabilities

144def random_choice_ll(judgments, judgment_repeats): 

145 """ 

146 Compute the log-likelihood under a random-choice benchmark. 

147 

148 This benchmark assumes every comparison direction is equally likely, so each 

149 observed judgment has probability 0.5. 

150 

151 Args: 

152 judgments (dict): Mapping from comparison keys to observed judgment counts. 

153 judgment_repeats (dict): Mapping from comparison keys to total repeats. 

154 

155 Returns: 

156 tuple[float, bool]: The log-likelihood and infeasibility flag returned by `calculate_ll`. 

157 """ 

158 counts = [] 

159 probs = [] 

160 repeats = [] 

161 for (k, v) in judgments.items(): 

162 counts.append(v) 

163 probs.append(0.5) 

164 repeats.append(judgment_repeats[k]) 

165 return calculate_ll(np.array(counts), np.array(probs), np.array(repeats))

168def best_model_ll(judgments, judgment_repeats): 

169 """ 

170 Compute the log-likelihood under an empirical best-case benchmark. 

171 

172 This benchmark assigns each comparison the probability estimated from its 

173 observed frequency in the data. 

174 

175 Args: 

176 judgments (dict): Mapping from comparison keys to observed judgment counts. 

177 judgment_repeats (dict): Mapping from comparison keys to total repeats. 

178 

179 Returns: 

180 tuple[float, bool]: The log-likelihood and infeasibility flag returned by `calculate_ll`. 

181 """ 

182 counts = [] 

183 probs = [] 

184 repeats = [] 

185 for (k, v) in judgments.items(): 

186 counts.append(v) 

187 probs.append(v / judgment_repeats[k]) 

188 repeats.append(judgment_repeats[k]) 

189 return calculate_ll(np.array(counts), np.array(probs), np.array(repeats))
